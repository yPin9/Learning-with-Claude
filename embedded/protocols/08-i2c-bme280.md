# Ch 8 — 實作：I2C → BME280 溫濕壓感測器

> 目標：用 Ch 7 的 register-level I2C driver 完整驅動 BME280（Bosch 溫濕壓感測器），涵蓋初始化、calibration data 讀取、burst read 資料，以及 Bosch 原廠補償公式的實作。

## BME280 硬體基本資訊

| 項目 | 說明 |
|------|------|
| I2C 位址 | SDO 接 GND → 0x76，SDO 接 VCC → 0x77 |
| VDD | 1.71 ~ 3.6V（3.3V 系統直接接，不用降壓） |
| 最大速率 | Fast Mode 400 kHz |
| Chip ID 暫存器 | 0xD0，值固定為 0x60 |
| Reset 暫存器 | 0xE0，寫入 0xB6 執行 soft reset |

SDO 接錯是最常見的接線錯誤：如果讀 chip_id 拿到 0xFF，先確認位址對不對。

## 關鍵暫存器地圖

```
位址    名稱                說明
------  ------------------  ------------------------------------------
0x88    calib00             Calibration data 起始（dig_T1 LSB）
 ..
0xA1    calib25             最後一個溫度/壓力校準 byte
0xD0    id                  Chip ID（應為 0x60）
0xE0    reset               寫 0xB6 觸發 soft reset
0xF3    status              bit3=measuring，bit0=im_update（NVRAM 複製中）
0xF4    ctrl_meas           [7:5]=osrs_t，[4:2]=osrs_p，[1:0]=mode
0xF5    config              [7:5]=t_sb，[4:2]=filter，[0]=spi3w_en
0xF7    press_msb           壓力資料 MSB
0xF8    press_lsb           壓力資料 LSB
0xF9    press_xlsb          [7:4]=壓力資料 XLSB
0xFA    temp_msb            溫度資料 MSB
0xFB    temp_lsb            溫度資料 LSB
0xFC    temp_xlsb           [7:4]=溫度資料 XLSB
0xFE    hum_msb             濕度資料 MSB（BME280 才有，BMP280 無）
0xFF    hum_lsb             濕度資料 LSB
```

本章只處理溫度和壓力（humidity 的補償公式更複雜，留給讀者自行加）。

## ctrl_meas 暫存器

```
0xF4 ctrl_meas:
  [7:5]  osrs_t  過取樣率（temperature）
  [4:2]  osrs_p  過取樣率（pressure）
  [1:0]  mode    00=sleep, 01/10=forced, 11=normal

osrs_t / osrs_p:
  000 = skipped（output = 0x80000）
  001 = x1
  010 = x2
  011 = x4
  100 = x8
  101+ = x16
```

設定 `osrs_t=010（x2）`、`osrs_p=101（x16）`、`mode=11（normal）`：

```
0xF4 = (0b010 << 5) | (0b101 << 2) | 0b11 = 0x57
```

## Calibration Data 結構

BME280 的 ADC 輸出是原始數值，必須用儲存在 NVRAM 的校準係數做補償才能得到真正的溫度和壓力。

溫度校準：3 個 16-bit 係數（dig_T1~T3），位於 0x88~0x8D。
壓力校準：9 個 16-bit 係數（dig_P1~P9），位於 0x8E~0x9F。

```c
typedef struct {
    // 溫度校準
    uint16_t dig_T1;
    int16_t  dig_T2;
    int16_t  dig_T3;
    // 壓力校準
    uint16_t dig_P1;
    int16_t  dig_P2;
    int16_t  dig_P3;
    int16_t  dig_P4;
    int16_t  dig_P5;
    int16_t  dig_P6;
    int16_t  dig_P7;
    int16_t  dig_P8;
    int16_t  dig_P9;
} bme280_calib_t;
```

所有係數都是 little-endian，16-bit，從 0x88 連續排列（可以一次 burst read 24 bytes）。

## I2C Register 讀寫輔助函式

先建立兩個基礎函式。這裡假設 Ch 7 的 `i2c_master_init()` 已經呼叫完畢。

```c
#include <stdint.h>
#include <stddef.h>
#include <string.h>

// 這兩個函式使用 Ch 7 的 i2c_write()，並補充 i2c_read()

// base address getter 已在 Ch 7 定義，這裡直接使用
extern uint32_t i2c_base_addr(int port);

#define REG_WRITE(addr, val)  (*((volatile uint32_t *)(addr)) = (val))
#define REG_READ(addr)        (*((volatile uint32_t *)(addr)))
#define REG_SET_BIT(addr, b)  REG_WRITE((addr), REG_READ(addr) | (b))
#define REG_CLR_BIT(addr, b)  REG_WRITE((addr), REG_READ(addr) & ~(b))

// 暫存器 offset（重複定義方便獨立編譯）
#define I2C_INT_CLR_REG    0x24
#define I2C_INT_RAW_REG    0x20
#define I2C_DATA_APB_REG   0x1C
#define I2C_FIFO_CONF_REG  0x18
#define I2C_CTR_REG        0x04
#define I2C_COMD0_REG      0x58

#define I2C_TRANS_COMPLETE_INT (1 << 7)
#define I2C_ACK_ERR_INT        (1 << 10)

#define I2C_CMD(op, byte_num, ack_en, ack_exp, ack_val) \
    (((op) << 11) | ((byte_num) & 0xFF) | \
     ((ack_en) << 8) | ((ack_exp) << 9) | ((ack_val) << 10))
#define CMD_RSTART 0
#define CMD_WRITE  1
#define CMD_READ   2
#define CMD_STOP   3

extern int i2c_write(int port, uint8_t slave_addr,
                     const uint8_t *data, size_t len);

// 寫 1 個暫存器位址，然後 Repeated START 讀 len 個 bytes
int i2c_read_reg(int port, uint8_t slave_addr,
                 uint8_t reg_addr, uint8_t *buf, size_t len) {
    uint32_t base = i2c_base_addr(port);
    size_t i;

    REG_WRITE(base + I2C_INT_CLR_REG, 0xFFFF);
    // 重置 FIFO
    REG_SET_BIT(base + I2C_FIFO_CONF_REG, (1 << 12) | (1 << 13));
    REG_CLR_BIT(base + I2C_FIFO_CONF_REG, (1 << 12) | (1 << 13));

    // TX FIFO：[ADDR+W] [REG_ADDR] [ADDR+R]
    REG_WRITE(base + I2C_DATA_APB_REG, (slave_addr << 1) | 0);  // ADDR+W
    REG_WRITE(base + I2C_DATA_APB_REG, reg_addr);                // 暫存器位址
    REG_WRITE(base + I2C_DATA_APB_REG, (slave_addr << 1) | 1);  // ADDR+R

    volatile uint32_t *comd = (volatile uint32_t *)(base + I2C_COMD0_REG);
    comd[0] = I2C_CMD(CMD_RSTART, 0, 0, 0, 0);              // START
    comd[1] = I2C_CMD(CMD_WRITE,  1, 1, 0, 0);              // ADDR+W
    comd[2] = I2C_CMD(CMD_WRITE,  1, 1, 0, 0);              // reg addr
    comd[3] = I2C_CMD(CMD_RSTART, 0, 0, 0, 0);              // Repeated START
    comd[4] = I2C_CMD(CMD_WRITE,  1, 1, 0, 0);              // ADDR+R
    // 讀 len-1 個 byte，每個送 ACK（ack_val=0）
    if (len > 1) {
        comd[5] = I2C_CMD(CMD_READ, (int)(len - 1), 1, 0, 0);  // ACK
    }
    // 最後 1 個 byte 送 NACK（ack_val=1，通知 slave 停止）
    comd[(len > 1) ? 6 : 5] = I2C_CMD(CMD_READ, 1, 1, 0, 1);  // NACK
    comd[(len > 1) ? 7 : 6] = I2C_CMD(CMD_STOP, 0, 0, 0, 0);  // STOP

    REG_SET_BIT(base + I2C_CTR_REG, (1 << 6));

    uint32_t timeout = 100000;
    uint32_t status;
    while (timeout--) {
        status = REG_READ(base + I2C_INT_RAW_REG);
        if (status & I2C_TRANS_COMPLETE_INT) break;
        if (status & I2C_ACK_ERR_INT) {
            REG_WRITE(base + I2C_INT_CLR_REG, 0xFFFF);
            return -1;
        }
    }
    if (!timeout) return -2;
    REG_WRITE(base + I2C_INT_CLR_REG, 0xFFFF);

    // 從 RX FIFO 取出資料
    for (i = 0; i < len; i++) {
        buf[i] = (uint8_t)(REG_READ(base + I2C_DATA_APB_REG) & 0xFF);
    }
    return 0;
}

static int i2c_write_reg(int port, uint8_t slave_addr,
                         uint8_t reg_addr, uint8_t value) {
    uint8_t buf[2] = { reg_addr, value };
    return i2c_write(port, slave_addr, buf, 2);
}
```

## BME280 完整驅動

```c
#define BME280_I2C_PORT    0
#define BME280_I2C_ADDR    0x76   // SDO=GND；若 SDO=VCC 改為 0x77

#define BME280_REG_ID       0xD0
#define BME280_REG_RESET    0xE0
#define BME280_REG_STATUS   0xF3
#define BME280_REG_CTRL_MEAS 0xF4
#define BME280_REG_CONFIG   0xF5
#define BME280_REG_PRESS_MSB 0xF7
#define BME280_CALIB_BASE   0x88

static bme280_calib_t g_calib;

// 從小端 byte array 組 uint16_t / int16_t
static uint16_t le16u(const uint8_t *p) { return (uint16_t)(p[0] | (p[1] << 8)); }
static int16_t  le16s(const uint8_t *p) { return (int16_t)(p[0] | (p[1] << 8)); }

static void bme280_read_calib(void) {
    uint8_t raw[26];
    i2c_read_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                 BME280_CALIB_BASE, raw, 26);
    g_calib.dig_T1 = le16u(raw + 0);
    g_calib.dig_T2 = le16s(raw + 2);
    g_calib.dig_T3 = le16s(raw + 4);
    g_calib.dig_P1 = le16u(raw + 6);
    g_calib.dig_P2 = le16s(raw + 8);
    g_calib.dig_P3 = le16s(raw + 10);
    g_calib.dig_P4 = le16s(raw + 12);
    g_calib.dig_P5 = le16s(raw + 14);
    g_calib.dig_P6 = le16s(raw + 16);
    g_calib.dig_P7 = le16s(raw + 18);
    g_calib.dig_P8 = le16s(raw + 20);
    g_calib.dig_P9 = le16s(raw + 22);
}

// 回傳 0=成功, 負數=失敗
int bme280_init(void) {
    uint8_t id = 0;
    int ret;

    // 1. Soft reset
    i2c_write_reg(BME280_I2C_PORT, BME280_I2C_ADDR, BME280_REG_RESET, 0xB6);
    // reset 需要約 2 ms
    volatile uint32_t d = 200000; while (d--) {};

    // 2. 驗證 chip ID
    ret = i2c_read_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                       BME280_REG_ID, &id, 1);
    if (ret != 0 || id != 0x60) return -1;

    // 3. 等待 NVRAM 複製完成（status.im_update=0）
    uint8_t status;
    uint32_t timeout = 10000;
    do {
        i2c_read_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                     BME280_REG_STATUS, &status, 1);
    } while ((status & 0x01) && --timeout);
    if (!timeout) return -2;

    // 4. 讀取 calibration data
    bme280_read_calib();

    // 5. 設定 oversampling：osrs_t=x2, osrs_p=x16
    //    config：t_sb=0.5ms, filter=x16
    i2c_write_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                  BME280_REG_CONFIG, 0x10);   // filter=x16

    // 6. 設 normal mode（osrs_t=010, osrs_p=101, mode=11 = 0x57）
    i2c_write_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                  BME280_REG_CTRL_MEAS, 0x57);
    return 0;
}

// Bosch 原廠溫度補償公式（datasheet 4.2.3）
// 輸入：raw ADC 值（20-bit，取自暫存器左移後結果）
// 輸出：溫度 × 100（整數攝氏），同時更新 t_fine（供壓力補償用）
static int32_t g_t_fine;

static int32_t bme280_compensate_temp(int32_t adc_T) {
    int32_t var1, var2, T;
    var1 = ((((adc_T >> 3) - ((int32_t)g_calib.dig_T1 << 1))) *
            ((int32_t)g_calib.dig_T2)) >> 11;
    var2 = (((((adc_T >> 4) - ((int32_t)g_calib.dig_T1)) *
              ((adc_T >> 4) - ((int32_t)g_calib.dig_T1))) >> 12) *
            ((int32_t)g_calib.dig_T3)) >> 14;
    g_t_fine = var1 + var2;
    T = (g_t_fine * 5 + 128) >> 8;
    return T;   // 單位：0.01 °C，例如 2345 = 23.45°C
}

// Bosch 原廠壓力補償公式（datasheet 4.2.3）
// 輸出：Pa × 256（Q24.8 格式），除以 256 得到 Pa
static uint32_t bme280_compensate_press(int32_t adc_P) {
    int64_t var1, var2, p;
    var1 = ((int64_t)g_t_fine) - 128000;
    var2 = var1 * var1 * (int64_t)g_calib.dig_P6;
    var2 = var2 + ((var1 * (int64_t)g_calib.dig_P5) << 17);
    var2 = var2 + (((int64_t)g_calib.dig_P4) << 35);
    var1 = ((var1 * var1 * (int64_t)g_calib.dig_P3) >> 8) +
           ((var1 * (int64_t)g_calib.dig_P2) << 12);
    var1 = (((((int64_t)1) << 47) + var1)) *
           ((int64_t)g_calib.dig_P1) >> 33;
    if (var1 == 0) return 0;
    p = 1048576 - adc_P;
    p = (((p << 31) - var2) * 3125) / var1;
    var1 = (((int64_t)g_calib.dig_P9) * (p >> 13) * (p >> 13)) >> 25;
    var2 = (((int64_t)g_calib.dig_P8) * p) >> 19;
    p = ((p + var1 + var2) >> 8) + (((int64_t)g_calib.dig_P7) << 4);
    return (uint32_t)p;   // Pa × 256
}

// 讀一筆溫度（單位 0.01°C）
// 同時更新 t_fine，下一次呼叫 bme280_read_pressure() 才有效
int32_t bme280_read_temperature(void) {
    uint8_t raw[6];
    // Burst read 0xF7 ~ 0xFC（press 3B + temp 3B）
    i2c_read_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                 BME280_REG_PRESS_MSB, raw, 6);
    int32_t adc_T = ((int32_t)raw[3] << 12) |
                    ((int32_t)raw[4] << 4)  |
                    (raw[5] >> 4);
    return bme280_compensate_temp(adc_T);
}

// 讀一筆壓力（Pa，整數）
uint32_t bme280_read_pressure(void) {
    uint8_t raw[6];
    i2c_read_reg(BME280_I2C_PORT, BME280_I2C_ADDR,
                 BME280_REG_PRESS_MSB, raw, 6);
    int32_t adc_P = ((int32_t)raw[0] << 12) |
                    ((int32_t)raw[1] << 4)  |
                    (raw[2] >> 4);
    int32_t adc_T = ((int32_t)raw[3] << 12) |
                    ((int32_t)raw[4] << 4)  |
                    (raw[5] >> 4);
    bme280_compensate_temp(adc_T);   // 更新 t_fine
    return bme280_compensate_press(adc_P) / 256;  // 轉回 Pa
}
```

## 邏輯分析儀驗證

用 Saleae 搭配 I2C decoder，驗證一筆 `bme280_read_temperature()` 的波形：

```
預期解碼輸出：
Transaction 1（write reg addr）：
  [START]
  [ADDR 0x76 + W] ACK
  [DATA 0xF7]     ACK
  [Repeated START]
  [ADDR 0x76 + R] ACK
  [DATA press_msb] ACK
  [DATA press_lsb] ACK
  [DATA press_xlsb] ACK
  [DATA temp_msb]  ACK
  [DATA temp_lsb]  ACK
  [DATA temp_xlsb] NACK   ← master 送 NACK 結束 read
  [STOP]
```

常見問題排查：

| 症狀 | 原因 |
|------|------|
| ADDR byte 收到 NACK | 位址錯（SDO 接線弄反），或 I2C 未初始化 |
| chip_id 讀到 0x00 或 0xFF | SDA/SCL 接反，或上拉電阻不在線 |
| 溫度值明顯偏差 | calibration data 讀取失敗（di_T1=0 是典型症狀） |
| 壓力=0 | t_fine 未更新（需先呼叫 read_temperature 或在 read_pressure 內呼叫補償） |
