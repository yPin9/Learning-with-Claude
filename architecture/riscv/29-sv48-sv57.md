# Ch 29 — Sv48 / Sv57：更大的地址空間，手動建立 Sv48 頁表

> 目標：理解 Sv48 比 Sv39 多了什麼；能手動寫一個 Sv48 四層頁表的 baremetal C 程式並在 QEMU 上跑通；知道常見的錯誤在哪裡。

---

## 29.1 Sv48 vs Sv39 vs Sv57 對照

| 模式  | VA 位元 | PA 位元 | 頁表層數 | 最大使用者 VA | MODE 值 | Linux 預設 |
|------|--------|--------|---------|------------|--------|-----------|
| Sv39 | 39     | 56     | 3       | 256 GiB    | 9      | kernel < 5.17 |
| Sv48 | 48     | 56     | 4       | 128 TiB    | 10     | kernel >= 5.17 |
| Sv57 | 57     | 56     | 5       | 64 PiB     | 11     | 實驗性，尚未預設 |

---

## 29.2 Sv48 的 VA 分解

```
Sv48 虛擬地址（64-bit 暫存器，低 48 bits 有效）：

  63     48 47   39 38   30 29   21 20   12 11        0
  +---------+-------+-------+-------+-------+-----------+
  |  sext48 |VPN[3] |VPN[2] |VPN[1] |VPN[0] |  offset   |
  +---------+-------+-------+-------+-------+-----------+
              9 bits  9 bits  9 bits  9 bits   12 bits

VPN[3]：root（L3）page table 的索引
VPN[2]：L2 page table 的索引
VPN[1]：L1 page table 的索引
VPN[0]：L0（leaf）page table 的索引
```

和 Sv39 相比，唯一的差別是多一個 VPN[3]——也就是多一層頁表。PTE 格式完全相同。

---

## 29.3 Sv48 頁表走法

```
satp.PPN × 4096
      |
      v
 Root Page Table（L3，512 個 PTE）
 index = VPN[3]
      |
      +--→ L2 Page Table（512 個 PTE）
            index = VPN[2]
                  |
                  +--→ L1 Page Table（512 個 PTE）
                        index = VPN[1]
                              |
                              +--→ L0（leaf，512 個 PTE）
                                    index = VPN[0]
                                          |
                                          +--→ PPN × 4096 + offset = PA
```

每層各佔 1 個 4 KiB page。完整 Sv48 頁表（全填滿）要 4 層 × 每層一個 page，加上更多的中間 page——實際 kernel 不會全填，只填有 mapping 的部分。

---

## 29.4 Sv48 的大頁

```
Megapage  （2 MiB）：在 L0 層放 leaf PTE（就是普通 page）
Megapage  （2 MiB）：在 L1 層放 leaf PTE
Gigapage  （1 GiB）：在 L2 層放 leaf PTE
Terapage  （512 GiB）：在 L3（root）層放 leaf PTE
             PTE.PPN[2:0] 必須是 0（對齊 512 GiB）
```

Terapage 在 Sv48 才有（Sv39 的 root 就是 L2，沒有 L3）。

---

## 29.5 手動建立 Sv48 頁表（完整 C Code）

這個程式在 QEMU virt 上跑，identity map 一個 page，然後通過虛擬地址讀取驗證。

```c
// sv48_baremetal.c
// 編譯：riscv64-unknown-elf-gcc -nostdlib -nostartfiles -T link.ld sv48_baremetal.c -o sv48.elf
// 執行：qemu-system-riscv64 -M virt -bios none -kernel sv48.elf -nographic

#include <stdint.h>

// ============================================================
// PTE flag 定義
// ============================================================
#define PTE_V  (1ULL << 0)   // Valid
#define PTE_R  (1ULL << 1)   // Read
#define PTE_W  (1ULL << 2)   // Write
#define PTE_X  (1ULL << 3)   // Execute
#define PTE_U  (1ULL << 4)   // User
#define PTE_G  (1ULL << 5)   // Global
#define PTE_A  (1ULL << 6)   // Accessed
#define PTE_D  (1ULL << 7)   // Dirty

// 把物理地址轉成 PTE 的 PPN 欄位
// PA >> 12 = PPN（物理頁號）
// PPN 放在 PTE bits [53:10]，所以 PPN << 10
#define MAKE_PTE_PPN(pa)  (((uint64_t)(pa) >> 12) << 10)

// 從 PTE 取回 physical address（下一層頁表的位址）
#define PTE_TO_PA(pte)    (((pte) >> 10) << 12)

// ============================================================
// 靜態分配 5 個 4KiB-aligned 頁面：
//   root_pt：L3 root
//   l2_pt  ：L2
//   l1_pt  ：L1
//   l0_pt  ：L0（leaf）
//   data_pt：我們要 map 的 data page（用來驗證）
// ============================================================
static uint64_t root_pt[512] __attribute__((aligned(4096)));
static uint64_t l2_pt  [512] __attribute__((aligned(4096)));
static uint64_t l1_pt  [512] __attribute__((aligned(4096)));
static uint64_t l0_pt  [512] __attribute__((aligned(4096)));

// data page 放一個已知的值
volatile uint64_t data_page[512] __attribute__((aligned(4096)));

// ============================================================
// UART 輸出（QEMU virt 的 UART 在 0x10000000）
// ============================================================
#define UART_BASE 0x10000000UL
static inline void uart_putc(char c) {
    *(volatile char *)UART_BASE = c;
}
static void uart_puts(const char *s) {
    while (*s) uart_putc(*s++);
}
static void uart_put_hex64(uint64_t val) {
    uart_puts("0x");
    for (int i = 60; i >= 0; i -= 4) {
        uint8_t nibble = (val >> i) & 0xF;
        uart_putc(nibble < 10 ? '0' + nibble : 'a' + nibble - 10);
    }
}

// ============================================================
// 設定 Sv48 頁表並切換
// ============================================================
void setup_sv48_pagetable(void) {
    // 我們要 identity-map 的 VA = PA = data_page 的地址
    uint64_t va = (uint64_t)data_page;
    uint64_t pa = va;  // identity map

    // 分解 VA
    uint64_t vpn3 = (va >> 39) & 0x1FF;
    uint64_t vpn2 = (va >> 30) & 0x1FF;
    uint64_t vpn1 = (va >> 21) & 0x1FF;
    uint64_t vpn0 = (va >> 12) & 0x1FF;

    uart_puts("VA = ");     uart_put_hex64(va);     uart_putc('\n');
    uart_puts("VPN[3] = "); uart_put_hex64(vpn3);   uart_putc('\n');
    uart_puts("VPN[2] = "); uart_put_hex64(vpn2);   uart_putc('\n');
    uart_puts("VPN[1] = "); uart_put_hex64(vpn1);   uart_putc('\n');
    uart_puts("VPN[0] = "); uart_put_hex64(vpn0);   uart_putc('\n');

    // 把一個已知值寫入 data page
    data_page[0] = 0xDEADBEEFCAFEBABEULL;

    // 建立四層頁表
    // L3 root → L2
    root_pt[vpn3] = MAKE_PTE_PPN((uint64_t)l2_pt) | PTE_V;
    // L2 → L1
    l2_pt[vpn2]   = MAKE_PTE_PPN((uint64_t)l1_pt) | PTE_V;
    // L1 → L0
    l1_pt[vpn1]   = MAKE_PTE_PPN((uint64_t)l0_pt) | PTE_V;
    // L0（leaf）→ data page
    l0_pt[vpn0]   = MAKE_PTE_PPN(pa) | PTE_V | PTE_R | PTE_W | PTE_A | PTE_D;

    // 設定 satp：MODE=10(Sv48), ASID=0, PPN=root_pt
    uint64_t satp_val = (10ULL << 60)                  // MODE = Sv48
                       | (0ULL  << 44)                 // ASID = 0
                       | ((uint64_t)root_pt >> 12);    // PPN

    uart_puts("satp = "); uart_put_hex64(satp_val); uart_putc('\n');

    // 切換！（注意：這條指令執行完下一條指令就在 Sv48 VA 空間裡了）
    __asm__ volatile (
        "csrw satp, %0\n\t"
        "sfence.vma zero, zero\n\t"  // 刷新 TLB
        :: "r"(satp_val) : "memory"
    );
}

// ============================================================
// 驗證：通過虛擬地址讀取 data_page[0]
// ============================================================
void verify_pagetable(void) {
    uint64_t *va_ptr = (uint64_t *)data_page;
    uint64_t val = *va_ptr;   // 如果頁表正確，這裡讀到 0xDEADBEEFCAFEBABE

    uart_puts("Read via VA: ");
    uart_put_hex64(val);
    uart_putc('\n');

    if (val == 0xDEADBEEFCAFEBABEULL) {
        uart_puts("PASS: Sv48 page table works!\n");
    } else {
        uart_puts("FAIL: wrong value\n");
    }
}

// ============================================================
// Entry point
// ============================================================
void _start(void) {
    uart_puts("=== Sv48 Page Table Demo ===\n");
    setup_sv48_pagetable();
    verify_pagetable();

    // 停在這裡
    while (1) {
        __asm__ volatile ("wfi");
    }
}
```

**link.ld（最小 linker script）：**

```ld
ENTRY(_start)
SECTIONS {
    . = 0x80200000;
    .text   : { *(.text*) }
    .rodata : { *(.rodata*) }
    .data   : { *(.data*) }
    .bss    : { *(.bss*) *(COMMON) }
}
```

**編譯與執行：**

```bash
riscv64-unknown-elf-gcc -nostdlib -nostartfiles -T link.ld \
    -O0 -g sv48_baremetal.c -o sv48.elf

qemu-system-riscv64 -M virt -bios none -kernel sv48.elf \
    -nographic -serial stdio
```

## 29.6 常見錯誤與除錯

**錯誤 1：PPN 計算錯誤**

```c
// 錯誤！直接把 PA 填進去
root_pt[vpn3] = (uint64_t)l2_pt | PTE_V;

// 正確：PA 要右移 12（去掉 page offset），再左移 10（對齊到 PTE 的 PPN 欄位起始位）
root_pt[vpn3] = MAKE_PTE_PPN((uint64_t)l2_pt) | PTE_V;
```

**錯誤 2：Non-leaf PTE 設了 R/W/X**

Non-leaf PTE 必須 R=W=X=0。如果你把中間層頁表的 PTE 設了 R 或 W，硬體會把它當成 leaf——但 PPN 指向的是另一張頁表，不是 data page，讀出來是垃圾。

**錯誤 3：satp 切換後下一條指令 page fault**

如果程式本身（.text 段）沒有被 map 到頁表裡，`csrw satp` 執行完，下一條 PC 對應的 VA 翻譯失敗，直接 instruction page fault。因為在 baremetal 環境裡沒有 exception handler，QEMU 會顯示類似：

```
riscv_cpu_do_interrupt: hart:0, async:0, cause:000...000c, epc:0x8020xxxx, tval:0x8020xxxx
```

解法：先把 .text 和 .data 也 identity-map 起來，再切換 satp。

**錯誤 4：沒有 sfence.vma**

`csrw satp` 只改了 satp CSR，但 TLB 還快取著舊的（或沒有）翻譯。沒有 `sfence.vma` 的話，後續存取的行為是 implementation-defined（QEMU 通常還是會重新 walk，但真實硬體不保證）。

## 29.7 Linux 用哪種模式

- Linux 5.17 以前：RV64 預設 Sv39
- Linux 5.17 起：RV64 預設 Sv48（如果硬體支援）
- 偵測方式：kernel 試著用 Sv48，如果 `csrw satp` 設了 MODE=10 之後讀回來 MODE 還是 10，就是支援

```
/proc/cpuinfo 的 mmu 欄位：
  mmu  : sv48
```

Sv57 目前只在 QEMU 和少數硬體支援，Linux 也有支援但不是預設。
